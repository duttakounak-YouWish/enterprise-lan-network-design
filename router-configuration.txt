! Router: Cisco 2911
! Hostname: R1
! LAN connection: R1 G0/0 -> SW1 G0/1
! LAN: 192.168.10.0/24

enable
configure terminal
hostname R1

interface GigabitEthernet0/0
 description CONNECTION_TO_SW1
 ip address 192.168.10.1 255.255.255.0
 no shutdown
exit

end
copy running-config startup-config

! Verification:
show ip interface brief
show running-config
ping 192.168.10.2
ping 192.168.10.3
